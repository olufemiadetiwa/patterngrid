SELECT l.location, s.units AS sales, k.avail_pct, d.deliveries
FROM dim.location l
JOIN sales.sell_out s ON s.outlet_code = l.outlet_code
JOIN stock.availability k ON k.loc_code = l.stock_code
JOIN dist.drops d ON d.drop_code = l.drop_code
WHERE s.week = 36 AND k.week = 36 AND d.week = 36;
